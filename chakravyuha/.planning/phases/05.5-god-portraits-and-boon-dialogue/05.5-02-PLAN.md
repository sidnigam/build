// js/dialogue.js — 'use strict'; top-level `const DIALOGUE = { ... };` No DOM access at all.
DIALOGUE.KNOWN_WHEN = ['first','minMeetings','firstThisRun','lowHp','biome','justDefeated','holds',
  'lastDeathBiome','lastDeathBoss','lastRunWon','minDeaths','minWins','weapon'];
DIALOGUE.PRI = { first:100, duo:90, justDefeated:70, lastDeathBiome:60, lastDeathBoss:60, lastRunWon:60,
  lowHp:50, holds:40, weapon:35, biome:30, minWins:20, minDeaths:20, minMeetings:20, firstThisRun:10 };
DIALOGUE.CHANCE = { 100:1, 90:1, 70:1, 60:0.8, 50:0.8, 40:0.5, 35:0.5, 30:0.4, 20:0.35, 10:0.35, 0:1 };
DIALOGUE.LOW_HP = 0.35;
DIALOGUE.LINES = [ ... ];            // content, Task 2
DIALOGUE.LORE = { godId: { s, d } }; // codex paragraph per god, Task 2

portraitPath(id) -> 'assets/sprites/gods/' + id + '-portrait.png'
speakerName(s)   -> s === 'abhimanyu' ? 'Abhimanyu' : DATA.GODS[s].name
beats(line)      -> line.beats ? line.beats.map(b => ({s:b.s, t:b.t}))
                                : [{s:line.god, t:line.text}].concat(line.reply ? [{s:'abhimanyu', t:line.reply}] : [])
priOf(line)      -> line.pri ?? (line.duo ? 90 : max(PRI[k] for k in keys(line.when||{})), else 0)
matches(when, ctx) -> every key: first:           ctx.first === v
                                  minMeetings:     ctx.meetings >= v
                                  firstThisRun:    ctx.firstThisRun === v
                                  lowHp:           (ctx.hpFrac <= LOW_HP) === v
                                  biome:           ctx.biome === v
                                  justDefeated:    v === '*' ? !!ctx.justDefeated : ctx.justDefeated === v
                                  holds:           (ctx.holds[v] | 0) > 0
                                  lastDeathBiome/lastDeathBoss: '*' ? truthy : equal
                                  lastRunWon:      ctx.lastRunWon === v
                                  minDeaths/minWins: ctx.deaths/wins >= v
                                  weapon:          ctx.weapon === v
                                  unknown key:     false
mem(meta = META) -> normalizes meta.dialogue IN PLACE (creates it if missing/non-object) to
                    {met:{}, seen:{}, last:{}, lastId:null, runMet:{run:-1, gods:[]}, lastRun:null}; returns it
buildContext(godId, opts = {}) -> const game = opts.game || GAME, meta = opts.meta || META, m = DIALOGUE.mem(meta);
   P = game.player; runMetGods = (m.runMet.run === meta.runs) ? m.runMet.gods : [];
   firstThisRun = !runMetGods.includes(godId);
   fresh = m.lastRun && m.lastRun.run === meta.runs - 1 && firstThisRun;
   holds = {} counting each P.allBoons() entry's def.gods (duo) or [def.god];
   return { god:godId, meetings:m.met[godId]|0, first:(m.met[godId]|0) === 0, firstThisRun,
     hpFrac: P ? P.hp / Math.max(1, P.maxhp) : 1, biome: game.biome ? game.biome.id : null,
     justDefeated: (game.biomeIdx > 0 && game.roomIdx <= 1) ? DATA.BIOMES[game.biomeIdx - 1].boss : null,
     holds, duo: opts.duo ? opts.duo.id || opts.duo : null,
     lastDeathBiome: fresh && m.lastRun.result === 'death' ? m.lastRun.biome : null,
     lastDeathBoss:  fresh && m.lastRun.result === 'death' ? m.lastRun.boss : null,
     lastRunWon: !!(fresh && m.lastRun.result === 'victory'),
     deaths: meta.deaths|0, wins: meta.wins|0, weapon: P && P.weapon ? P.weapon.id : meta.lastWeapon }
select(godId, ctx, mem, rand = Math.random) -> line | null      // pure: never mutates mem
   elig = LINES.filter(l => (l.god === godId && !l.duo && matches(l.when||{}, ctx)) ||
                             (l.duo && l.duo === ctx.duo && duoGods(l.duo).includes(godId)))
               .filter(l => !(l.once && mem.seen[l.id]));
   if (!elig.length) return null;
   fresh = elig.filter(l => l.id !== mem.last[godId] && l.id !== mem.lastId); pool = fresh.length ? fresh : elig;
   for pri in distinct(pool.map(priOf)) sorted desc:
      cands = pool.filter(l => priOf(l) === pri && rand() < (l.chance ?? CHANCE[pri] ?? 1));
      if (cands.length) return weightedPick(cands, l => (l.w ?? 1) / (1 + (mem.seen[l.id]|0)), rand);
   return pool[0];   // fallback (unreachable when a pri-0 generic exists; keep for safety)
record(line, godId, m = DIALOGUE.mem(), meta = META) -> m.met[godId]=(m.met[godId]|0)+1; m.seen[line.id]=(|0)+1;
   m.last[godId]=line.id; m.lastId=line.id; if (m.runMet.run !== meta.runs) m.runMet={run:meta.runs,gods:[]};
   if (!m.runMet.gods.includes(godId)) m.runMet.gods.push(godId); if (meta.save) meta.save(); return m;
   (for duo lines, ALSO bump met/last for the partner god, so meeting counts reflect both speakers)
noteRunEnd({result, biome, boss, run}, meta = META) -> mem(meta).lastRun = {result, biome, boss:boss||null, run}; meta.save()
metGods(meta = META) -> Object.keys(DATA.GODS).filter(id => (mem(meta).met[id]|0) > 0)
validate(lines = LINES) -> string[]  (D-17 rules; see Task 1)
coverage(lines = LINES) -> string[]  (GOD-04 rules; see Task 2)
